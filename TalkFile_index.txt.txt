    <script>
        // 대여소 데이터
        let stations = [
            { id: 1, name: "형남공학관", count: 12 },
            { id: 2, name: "미래관", count: 8 },
            { id: 3, name: "학생회관", count: 15 }
        ];

        let isRenting = false;
        let currentRentedStation = "";
        let timerInterval = null;
        let selectedStationId = null;

        // 화면 렌더링
        function renderStations() {
            const listContainer = document.getElementById('station-list');
            listContainer.innerHTML = '';

            stations.forEach(station => {
                const card = document.createElement('div');
                card.className = 'station-card';
                card.innerHTML = `
                    <div class="station-info">
                        <h3>${station.name}</h3>
                        <p>대여 가능 수량: <strong>${station.count}대</strong></p>
                    </div>
                    <button class="btn" ${isRenting || station.count === 0 ? 'disabled' : ''} onclick="openPaymentModal(${station.id})">
                        ${station.count === 0 ? '재고 없음' : '대여하기'}
                    </button>
                `;
                listContainer.appendChild(card);
            });
        }

        // 결제 모달 열기
        function openPaymentModal(stationId) {
            selectedStationId = stationId;
            const station = stations.find(s => s.id === stationId);
            document.getElementById('selected-station-name').textContent = station.name;
            document.getElementById('payment-modal').classList.add('show');
        }

        // 모달 닫기
        function closeModal(modalId) {
            document.getElementById(modalId).classList.remove('show');
        }

        // 대여 완료 처리
        function completeRental() {
            closeModal('payment-modal');
            const station = stations.find(s => s.id === selectedStationId);
            station.count--; // 수량 차감
            
            isRenting = true;
            currentRentedStation = station.name;

            // UI 변경
            document.getElementById('active-rental-section').classList.add('show');
            document.getElementById('active-station-text').textContent = `대여 장소: ${currentRentedStation}`;
            
            renderStations();
            startTimer(72 * 3600); // 72시간 타이머 시작
            alert('대여가 완료되었습니다! 우산을 안전하게 이용해 주세요.');
        }

        // 72시간 타이머 구동
        function startTimer(duration) {
            let timer = duration;
            clearInterval(timerInterval);
            
            timerInterval = setInterval(() => {
                let hours = Math.floor(timer / 3600);
                let minutes = Math.floor((timer % 3600) / 60);
                let seconds = timer % 60;

                hours = hours < 10 ? "0" + hours : hours;
                minutes = minutes < 10 ? "0" + minutes : minutes;
                seconds = seconds < 10 ? "0" + seconds : seconds;

                document.getElementById('timer-display').textContent = `${hours}:${minutes}:${seconds}`;

                if (--timer < 0) {
                    clearInterval(timerInterval);
                    alert('대여 시간이 만료되었습니다. 추가 요금이 부과될 수 있습니다.');
                }
            }, 1000);
        }

        // 반납하기 프로세스
        function openReturnModal() {
            // 어느 곳에 반납할지 선택지를 포함한 프롬프트 또는 간이 처리
            let returnLocation = prompt("반납하실 장소를 선택해 주세요.\n1. 형남공학관\n2. 미래관\n3. 학생회관\n(숫자 또는 장소명을 입력하세요)");
            
            let matchedStation = stations.find(s => s.name === returnLocation || s.id == returnLocation);
            
            if (!matchedStation) {
                // 기본값으로 학생회관 지정 혹은 취소 시
                if(returnLocation !== null) {
                    matchedStation = stations[2]; // 기본 학생회관
                } else {
                    return;
                }
            }

            // 재고 증가
            matchedStation.count++;
            
            // 타이머 종료 및 상태 초기화
            clearInterval(timerInterval);
            isRenting = false;
            document.getElementById('active-rental-section').classList.remove('show');

            // 영수증 표시
            document.getElementById('receipt-start').textContent = currentRentedStation;
            document.getElementById('receipt-end').textContent = matchedStation.name;
            document.getElementById('receipt-modal').classList.add('show');

            renderStations();
        }

        // 초기 실행
        renderStations();
    </script>
